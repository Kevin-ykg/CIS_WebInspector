cmake_minimum_required(VERSION 3.20)
project(CISVisionCore VERSION 2.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
option(QR_READER_ONLY "Build the reusable QR reader without CIS alignment/defect modules" OFF)
option(QR_BUILD_EXAMPLES "Build the standalone QR image examples" OFF)
option(QR_BUILD_TESTS "Build developer-only QR contract checks" OFF)
set(QR_COMPONENTS core imgproc objdetect wechat_qrcode)
if(QR_BUILD_EXAMPLES OR QR_BUILD_TESTS)
  list(APPEND QR_COMPONENTS imgcodecs)
endif()
if(NOT QR_READER_ONLY)
  list(APPEND QR_COMPONENTS features2d calib3d ximgproc)
endif()
find_package(OpenCV 4.10.0 EXACT REQUIRED COMPONENTS ${QR_COMPONENTS})

# 算法源码只编译为一个共享的静态核心。CIS 适配层和独立读码器均链接它，不复制算法。
add_library(QrRecognition STATIC src/qr_detector.cpp src/qr_geometry.cpp src/qr_recovery.cpp src/qr_blur_local.cpp)
target_compile_features(QrRecognition PUBLIC cxx_std_17)
target_compile_definitions(QrRecognition PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(QrRecognition PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(QrRecognition PUBLIC opencv_core opencv_imgproc opencv_objdetect opencv_wechat_qrcode)
target_compile_options(QrRecognition PRIVATE /W4 /utf-8 /EHsc /fp:precise)

add_library(QrReader SHARED src/qr_reader_api.cpp)
target_link_libraries(QrReader PRIVATE QrRecognition)
target_compile_definitions(QrReader PRIVATE QR_READER_EXPORTS NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(QrReader PRIVATE /W4 /utf-8 /EHsc /fp:precise)

if(QR_BUILD_EXAMPLES)
  add_executable(QrReaderExample examples/qr_reader.cpp)
  target_link_libraries(QrReaderExample PRIVATE QrRecognition opencv_imgcodecs)
  target_compile_options(QrReaderExample PRIVATE /W4 /utf-8 /EHsc)
  add_executable(QrReaderCExample examples/qr_reader_c_api.cpp)
  target_include_directories(QrReaderCExample PRIVATE include ${OpenCV_INCLUDE_DIRS})
  target_compile_features(QrReaderCExample PRIVATE cxx_std_17)
  target_link_libraries(QrReaderCExample PRIVATE QrReader opencv_imgcodecs opencv_core)
  target_compile_options(QrReaderCExample PRIVATE /W4 /utf-8 /EHsc)
endif()
if(QR_BUILD_TESTS)
  add_executable(QrReaderRecoveryChecks tests/qr_recovery_checks.cpp)
  target_link_libraries(QrReaderRecoveryChecks PRIVATE QrRecognition opencv_imgcodecs)
  target_compile_options(QrReaderRecoveryChecks PRIVATE /W4 /utf-8 /EHsc)
  add_executable(QrReaderChecks tests/qr_reader_checks.cpp src/cis_qr_adapter.cpp)
  target_include_directories(QrReaderChecks PRIVATE src)
  target_link_libraries(QrReaderChecks PRIVATE QrRecognition QrReader opencv_imgcodecs)
  target_compile_options(QrReaderChecks PRIVATE /W4 /utf-8 /EHsc)
endif()

if(NOT QR_READER_ONLY)
add_library(CISVisionCore SHARED src/qr_api.cpp src/cis_qr_adapter.cpp
  src/patch_api.cpp src/patch_detector.cpp src/patch_alignment.cpp src/patch_fine_line.cpp
  src/alignment_api.cpp src/alignment_global.cpp src/alignment_marks.cpp
  src/alignment_side_grid.cpp src/alignment_white_ink.cpp src/alignment_warp.cpp)
target_compile_features(CISVisionCore PRIVATE cxx_std_17)
target_compile_definitions(CISVisionCore PRIVATE CIS_VISION_EXPORTS NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(CISVisionCore PUBLIC include PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(CISVisionCore PRIVATE QrRecognition ${OpenCV_LIBS})
target_compile_options(CISVisionCore PRIVATE /W4 /utf-8 /EHsc /fp:precise)
endif()
